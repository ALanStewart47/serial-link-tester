cmake_minimum_required(VERSION 3.19)
project(controller_network_diagnostics VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Network Widgets)
qt_standard_project_setup()

add_library(network_diagnostics_core STATIC core/NetworkPathCheck.cpp core/NetworkPathCheck.h core/AppIdentity.h)
target_sources(network_diagnostics_core PRIVATE core/ControllerDiscovery.cpp core/ControllerDiscovery.h
    core/TcpConnectionCheck.cpp core/TcpConnectionCheck.h core/DiagnosticCommands.cpp core/DiagnosticCommands.h
    core/LocalTcpConnection.cpp core/LocalTcpConnection.h resources/diagnostic_commands.qrc
    core/DiagnosticEngine.cpp core/DiagnosticEngine.h)
target_include_directories(network_diagnostics_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(network_diagnostics_core PUBLIC Qt6::Core Qt6::Network)
if(WIN32)
    target_compile_definitions(network_diagnostics_core PRIVATE _WIN32_WINNT=0x0600 WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(network_diagnostics_core PRIVATE iphlpapi ws2_32)
endif()

qt_add_executable(controller_network_diagnostics WIN32 main.cpp ui/DiagnosticWindow.cpp ui/DiagnosticWindow.h)
target_sources(controller_network_diagnostics PRIVATE ui/CommandSelectionDialog.cpp ui/CommandSelectionDialog.h)
target_link_libraries(controller_network_diagnostics PRIVATE network_diagnostics_core Qt6::Widgets)

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(network_diagnostics_tests tests/NetworkPathCheckTest.cpp ui/DiagnosticWindow.cpp ui/DiagnosticWindow.h)
    target_sources(network_diagnostics_tests PRIVATE ui/CommandSelectionDialog.cpp ui/CommandSelectionDialog.h)
    target_link_libraries(network_diagnostics_tests PRIVATE network_diagnostics_core Qt6::Test Qt6::Widgets)
    set_target_properties(network_diagnostics_tests PROPERTIES WIN32_EXECUTABLE FALSE)
    add_test(NAME network_diagnostics_tests COMMAND network_diagnostics_tests -platform offscreen)
    set_tests_properties(network_diagnostics_tests PROPERTIES TIMEOUT 30)
    qt_add_executable(connection_check_tests tests/ConnectionCheckTest.cpp ui/DiagnosticWindow.cpp ui/DiagnosticWindow.h)
    target_sources(connection_check_tests PRIVATE ui/CommandSelectionDialog.cpp ui/CommandSelectionDialog.h)
    target_link_libraries(connection_check_tests PRIVATE network_diagnostics_core Qt6::Test Qt6::Widgets)
    set_target_properties(connection_check_tests PROPERTIES WIN32_EXECUTABLE FALSE)
    add_test(NAME connection_check_tests COMMAND connection_check_tests -platform offscreen)
    set_tests_properties(connection_check_tests PROPERTIES TIMEOUT 30)
    qt_add_executable(diagnostic_command_tests tests/DiagnosticCommandsTest.cpp)
    target_sources(diagnostic_command_tests PRIVATE ui/CommandSelectionDialog.cpp ui/CommandSelectionDialog.h)
    target_link_libraries(diagnostic_command_tests PRIVATE network_diagnostics_core Qt6::Test Qt6::Widgets)
    set_target_properties(diagnostic_command_tests PROPERTIES WIN32_EXECUTABLE FALSE)
    add_test(NAME diagnostic_command_tests COMMAND diagnostic_command_tests -platform offscreen)
    set_tests_properties(diagnostic_command_tests PROPERTIES TIMEOUT 30)
    qt_add_executable(diagnostic_engine_tests tests/DiagnosticEngineTest.cpp)
    target_link_libraries(diagnostic_engine_tests PRIVATE network_diagnostics_core Qt6::Test)
    set_target_properties(diagnostic_engine_tests PROPERTIES WIN32_EXECUTABLE FALSE)
    add_test(NAME diagnostic_engine_tests COMMAND diagnostic_engine_tests)
    set_tests_properties(diagnostic_engine_tests PROPERTIES TIMEOUT 30)
endif()
